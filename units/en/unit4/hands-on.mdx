from smolagents import CodeAgent, InferenceClientModel, DuckDuckGoSearchTool, VisitWebpageTool

class BasicAgent:
    def __init__(self):
        model = InferenceClientModel(model_id="Qwen/Qwen2.5-Coder-32B-Instruct")
        self.agent = CodeAgent(
            tools=[DuckDuckGoSearchTool(), VisitWebpageTool()],
            model=model,
            additional_authorized_imports=["requests", "re", "json", "math", "datetime", "pandas"],
            max_steps=8,
        )

    def __call__(self, question: str) -> str:
        prompt = (
            "Answer the question below. Use the tools to search the web when needed. "
            "Your final answer must be ONLY the answer itself: no explanation, no sentence, "
            "no 'FINAL ANSWER' text, no trailing period. Numbers without units unless asked. "
            "Lists must be comma-separated.\n\n"
            f"Question: {question}"
        )
        try:
            return str(self.agent.run(prompt)).strip()
        except Exception as e:
            return f"error: {e}"
